html:

<button onclick="showToast()">Bấm vào đây</button>

<div id="toast">
    Đã lưu thành công! ✅
</div>

<script>
function showToast() {
    let toast = document.getElementById("toast");

    toast.classList.add("show");

    setTimeout(function() {
        toast.classList.remove("show");
    }, 3000);
}
</script>

css :

#toast {
    position: fixed;
    top: 50%;
    left: 50%;
    transform: translate(-50%, -50%);

    background-color: #1e293b;
    color: white;

    padding: 20px 30px;
    border-radius: 10px;

    opacity: 0;
    visibility: hidden;

    transition: 0.3s;
}

#toast.show {
    opacity: 1;
    visibility: visible;
}
